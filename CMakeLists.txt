cmake_minimum_required(VERSION 3.16)
project(OblivionConfigTool VERSION 1.0 LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This tool uses Windows DXGI and file APIs.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/presets/Engine.ini.in" ENGINE_PRESET)
configure_file(src/config_template.h.in generated/config_template.h @ONLY)
add_library(config-core src/config.cpp)
target_include_directories(config-core PUBLIC src PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(config-core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_executable(oblivion-config main.cpp src/gpu.cpp)
target_link_libraries(oblivion-config PRIVATE config-core dxgi)
if(MINGW)
    target_link_options(oblivion-config PRIVATE -municode)
endif()
foreach(target config-core oblivion-config)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
    add_executable(config-tests tests/config_tests.cpp)
    target_link_libraries(config-tests PRIVATE config-core)
    add_test(NAME config-regressions COMMAND config-tests)
endif()
